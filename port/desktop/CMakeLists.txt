cmake_minimum_required(VERSION 3.20)
project(sm64ds_native_desktop_host LANGUAGES C CXX)
if(NOT CMAKE_SYSTEM_NAME MATCHES "^(Linux|Darwin)$")
    message(FATAL_ERROR "This target verifies the native Linux/macOS desktop adapter, not Windows or mobile apps")
endif()
option(DESKTOP_BUNDLED_SDL2 "Build the verified SDL2 Classic dependency instead of a system compatibility shim" ON)
if(DESKTOP_BUNDLED_SDL2)
    include(FetchContent)
    set(SDL_SHARED OFF CACHE BOOL "" FORCE)
    set(SDL_STATIC ON CACHE BOOL "" FORCE)
    set(SDL_TEST OFF CACHE BOOL "" FORCE)
    set(SDL_TESTS OFF CACHE BOOL "" FORCE)
    FetchContent_Declare(desktop_sdl2
        URL https://github.com/libsdl-org/SDL/archive/refs/tags/release-2.32.10.tar.gz
        URL_HASH SHA256=03f9d7c191a837525c9cda6406af2f2e48be02b5e7eb03d949cc9f1e9ca41c8b)
    FetchContent_MakeAvailable(desktop_sdl2)
    set(DESKTOP_SDL SDL2::SDL2-static)
else()
    find_package(SDL2 CONFIG QUIET)
    if(TARGET SDL2::SDL2)
        set(DESKTOP_SDL SDL2::SDL2)
    else()
        find_package(PkgConfig REQUIRED)
        pkg_check_modules(SDL2 REQUIRED IMPORTED_TARGET sdl2)
        set(DESKTOP_SDL PkgConfig::SDL2)
    endif()
endif()
find_package(Threads REQUIRED)
add_library(sm64ds_desktop_host STATIC desktop_host.cpp desktop_storage.cpp)
target_compile_features(sm64ds_desktop_host PUBLIC cxx_std_17)
target_include_directories(sm64ds_desktop_host PUBLIC .)
target_link_libraries(sm64ds_desktop_host PUBLIC ${DESKTOP_SDL})
target_compile_definitions(sm64ds_desktop_host PUBLIC SDL_MAIN_HANDLED)
target_compile_options(sm64ds_desktop_host PRIVATE -Wall -Wextra -Werror)
add_executable(native_desktop_host_test ../tests/coop_native_desktop_host_test.cpp)
target_link_libraries(native_desktop_host_test PRIVATE sm64ds_desktop_host Threads::Threads)
target_compile_options(native_desktop_host_test PRIVATE -Wall -Wextra -Werror -UNDEBUG)
option(DESKTOP_HOST_SANITIZERS "Check native host memory/undefined behavior" OFF)
if(DESKTOP_HOST_SANITIZERS)
    target_compile_options(sm64ds_desktop_host PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_compile_options(native_desktop_host_test PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(native_desktop_host_test PRIVATE -fsanitize=address,undefined)
endif()
enable_testing()
add_test(NAME native_window_input_pixels_audio COMMAND native_desktop_host_test)
add_executable(native_desktop_storage_test ../tests/coop_native_desktop_storage_test.cpp)
target_link_libraries(native_desktop_storage_test PRIVATE sm64ds_desktop_host Threads::Threads)
target_compile_options(native_desktop_storage_test PRIVATE -Wall -Wextra -Werror -UNDEBUG)
if(DESKTOP_HOST_SANITIZERS)
    target_compile_options(native_desktop_storage_test PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(native_desktop_storage_test PRIVATE -fsanitize=address,undefined)
endif()
add_test(NAME native_atomic_save_storage COMMAND native_desktop_storage_test)
set_tests_properties(native_atomic_save_storage PROPERTIES TIMEOUT 30)
set_tests_properties(native_window_input_pixels_audio PROPERTIES
    ENVIRONMENT "SDL_VIDEODRIVER=dummy;SDL_AUDIODRIVER=dummy" TIMEOUT 30)
